WATCHLIST ENGINE const watchlists = { default: [ { type: 'section', title: '### STOCKS' }, { type: 'ticker', symbol: 'AAPL', name: 'Apple Inc.' }, { type: 'ticker', symbol: 'TSLA', name: 'Tesla Inc.' }, { type: 'ticker', symbol: 'NVDA', name: 'NVIDIA Corp.' }, { type: 'section', title: '### CRYPTO (YAHOO)' }, { type: 'ticker', symbol: 'BTC-USD', name: 'Bitcoin USD' }, { type: 'ticker', symbol: 'ETH-USD', name: 'Ethereum USD' } ] }; function renderWatchlist() { const container = document.getElementById('watchlist-items'); if (!container) return; container.innerHTML = ''; watchlists.default.forEach((item, idx) => { if (item.type === 'section') { const div = document.createElement('div'); div.className = 'text-[11px] font-bold text-gray-400 pt-3 pb-1 px-2 border-b border-[#2a2e39] flex justify-between items-center'; div.innerHTML = `${item.title}`; container.appendChild(div); } else { const div = document.createElement('div'); div.className = `p-2 rounded flex justify-between items-center cursor-pointer hover:bg-[#2a2e39] ${item.symbol === currentSymbol ? 'bg-[#2a2e39] border-l-2 border-blue-500' : ''}`; div.onclick = () => { currentSymbol = item.symbol; document.getElementById('symbol-display').innerText = item.symbol; loadData(item.symbol, currentTimeframe); renderWatchlist(); }; div.innerHTML = `
${item.symbol}
${item.name}
`; container.appendChild(div); } }); } function removeWatchlistItem(idx) { watchlists.default.splice(idx, 1); renderWatchlist(); } // Safely attach event listeners after DOM elements are fully loaded document.addEventListener('DOMContentLoaded', () => { initCharts(); renderWatchlist(); const symbolDisplay = document.getElementById('symbol-display'); if (symbolDisplay) symbolDisplay.innerText = currentSymbol; const addSectionBtn = document.getElementById('add-section-btn'); if (addSectionBtn) { addSectionBtn.onclick = () => { const title = prompt("Enter section name:", "### NEW SECTION"); if (title) { watchlists.default.push({ type: 'section', title: title.toUpperCase() }); renderWatchlist(); } }; } const addTickerBtn = document.getElementById('add-ticker-btn'); if (addTickerBtn) { addTickerBtn.onclick = () => { const sym = prompt("Enter Yahoo ticker symbol (e.g., AAPL, MSFT, BTC-USD):"); if (sym) { watchlists.default.push({ type: 'ticker', symbol: sym.toUpperCase(), name: sym.toUpperCase() }); renderWatchlist(); } }; } // Initial data fetch loadData(currentSymbol, currentTimeframe); });
